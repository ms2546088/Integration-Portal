🐳 Docker Container Logs 
# View container logs (run on host):
docker logs --tail 50 CONTAINER_NAME
docker logs -f CONTAINER_NAME        # follow (stream)
docker logs --since 1h CONTAINER_NAME # last 1 hour

# Common containers:
docker logs --tail 20 docker_forgejo
docker logs --tail 20 docker_portainer_docs
docker logs --tail 20 coolify
🖥️ System Logs 
# Systemd service logs:
journalctl --user -u webhook-receiver -n 50
journalctl --user -u deploy-api -n 50
journalctl --user -u docker-mgmt-api -n 50

# System logs:
sudo journalctl -u docker --since "1 hour ago"
sudo journalctl -u nginx --since "1 hour ago"
sudo journalctl -u postgresql@18-main -n 50
