🚀 One-Click Smart Update

⚠️ Read Before Update

Updating may overwrite custom files. Must do before update:

  1. Backup all custom files (Portal HTML/config/compose)
  2. Record current version
  3. Check if custom files overwritten after update
  4. If overwritten, restore from backup
📦 Step 1: Backup 
BACKUP=/tmp/backup-$(date +%Y%m%d-%H%M)
mkdir -p $BACKUP

# BackupCustom Files
cp -r ~/docker/_portainer/www/Developer-ms202603jpc/ $BACKUP/
cp ~/docker/ubuntu/port-registry.json $BACKUP/
cp ~/docker/ubuntu/docker-compose.yml $BACKUP/

# Backup Stack Config
for s in 14 15 16; do
  cp ~/docker/ubuntu/Official-Area_Portainer/data/portainer/compose/$s/.env $BACKUP/stack$s.env
done
🔄 Step 2: UpdateService 
cd ~/docker/ubuntu

# Core Services (Forgejo/Windmill/Woodpecker)
docker compose pull && docker compose up -d

# Portainer Stacks
for s in 14 15 16; do
  cd Official-Area_Portainer/data/portainer/compose/$s
  docker compose pull && docker compose up -d
  cd ~/docker/ubuntu
done

# Cleanup Old Images
docker image prune -f
✅ Step 3: Verify 
# CheckAllContainerStatus
docker ps --format 'table {{.Names}}\t{{.Status}}'

# Verify Services Reachable
for p in 8000 9000 53000 58090 58100 59009; do
  curl -so /dev/null -w "%{http_code} :$p\n" http://10.1.4.201:$p/
done

# Compare Custom Files
diff $BACKUP/Developer-ms202603jpc/index.html ~/docker/_portainer/www/Developer-ms202603jpc/index.html
