🚀 One-Click Smart Update

⚠️ Read Before Update

Updating may overwrite custom files. Steps enforced by the script:

  1. Auto-backup all compose/env/portal files before update
  2. Snapshot container state (names, images, health)
  3. Pull latest images + recreate only changed containers
  4. Verify container count matches pre-update state
  5. Restore on rollback
🚀 One-Click Update (Recommended) 
# Full update with automatic backup
bash /home/admin958168/_docker/ubuntu/one-click-update.sh

# Dry run (preview only, no changes)
bash /home/admin958168/_docker/ubuntu/one-click-update.sh --dry-run

# Skip backup (not recommended)
bash /home/admin958168/_docker/ubuntu/one-click-update.sh --no-backup

Script: /home/admin958168/_docker/ubuntu/one-click-update.sh

Logs: /home/admin958168/_docker/ubuntu/logs/update-*.log

Backups: /home/admin958168/_docker/ubuntu/backups/YYYYMMDD-HHMMSS/

🔧 Manual Update Steps 
# Step 1: Backup
BACKUP=~/docker/ubuntu/backups/$(date +%Y%m%d-%H%M)
mkdir -p $BACKUP
cp ~/docker/ubuntu/docker-compose.yml $BACKUP/
cp ~/docker/_portainer/.env $BACKUP/portainer-.env

# Step 2: Pull and recreate
cd ~/docker/ubuntu
docker compose pull && docker compose up -d
cd ~/docker/_portainer
docker compose up -d

# Step 3: Stacks
for s in 14 15 16; do
  cd ~/docker/ubuntu/Official-Area_Portainer/data/portainer/compose/$s
  docker compose pull && docker compose up -d
done

# Step 4: Cleanup
docker image prune -f
✅ Post-Update Verification 
# Container health check
docker ps --format 'table {{.Names}}\t{{.Status}}\t{{.Image}}'

# Service reachability
for p in 8000 9000 53000 58090 58100 59000 59009; do
  curl -so /dev/null -w "%{http_code} :$p\n" http://10.1.4.201:$p/
done

# Live status JSON
bash /home/admin958168/_docker/ubuntu/live-status.sh health

# Compare pre/post state
diff ~/docker/ubuntu/logs/pre-update-state.txt \
     ~/docker/ubuntu/logs/post-update-state.txt
⏪ Rollback (if update breaks something) 
# Find latest backup
ls -lt ~/docker/ubuntu/backups/ | head -5

# Restore compose files from backup
BACKUP=~/docker/ubuntu/backups/YYYYMMDD-HHMMSS
cp $BACKUP/ubuntu-compose.yml ~/docker/ubuntu/docker-compose.yml
cp $BACKUP/portainer-.env ~/docker/_portainer/.env

# Restart with restored config
cd ~/docker/ubuntu && docker compose up -d
cd ~/docker/_portainer && docker compose up -d
