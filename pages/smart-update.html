Update Procedure

Step 1: Pre-Update Backup (MANDATORY) 
# Backup all custom configs and portal files
BACKUP_DIR="/home/admin958168/_docker/backups/$(date +%Y%m%d_%H%M%S)"
mkdir -p "$BACKUP_DIR"

# Portal files
cp -r /home/admin958168/_docker/_portainer/www/Developer-ms202603jpc "$BACKUP_DIR/portal/"

# Custom API services
cp /home/admin958168/_docker/ubuntu/docker-mgmt-api.py "$BACKUP_DIR/"
cp /home/admin958168/_docker/ubuntu/deploy-api.py "$BACKUP_DIR/"
cp /home/admin958168/_docker/ubuntu/webhook-receiver.py "$BACKUP_DIR/"
cp /home/admin958168/_docker/ubuntu/webhook-deploy.sh "$BACKUP_DIR/"

# Docker compose files
cp -r /home/admin958168/_docker/ubuntu/compose* "$BACKUP_DIR/" 2>/dev/null

# Systemd services
cp ~/.config/systemd/user/*api*.service "$BACKUP_DIR/" 2>/dev/null
cp ~/.config/systemd/user/webhook*.service "$BACKUP_DIR/" 2>/dev/null

echo "Backup created: $BACKUP_DIR"
Step 2: Update Docker Images 
# Pull latest images for all containers
docker pull nginx:alpine
docker pull mysql:8.0
docker pull mcr.microsoft.com/mssql/server:2022-latest
docker pull gitea/gitea:latest    # Forgejo

# Recreate containers (compose stacks)
cd /home/admin958168/_docker/ubuntu
docker compose -f compose-forgejo.yml up -d
docker compose -f compose-windmill.yml up -d
docker compose -f compose-woodpecker.yml up -d
Step 3: System Updates 
# Ubuntu system
sudo apt update && sudo apt upgrade -y

# Docker Engine
sudo apt install --only-upgrade docker-ce docker-ce-cli

# PostgreSQL
sudo apt install --only-upgrade postgresql-18

# Node.js
sudo npm install -g n && sudo n lts

# Python packages
pip3 install --upgrade pip
Step 4: Post-Update Restore (if needed) 
# If custom files were overwritten, restore from backup:
BACKUP_DIR="/home/admin958168/_docker/backups/YYYYMMDD_HHMMSS"  # Use actual backup dir

# Restore portal
cp -r "$BACKUP_DIR/portal/"* /home/admin958168/_docker/_portainer/www/Developer-ms202603jpc/

# Restore API services
cp "$BACKUP_DIR/docker-mgmt-api.py" /home/admin958168/_docker/ubuntu/
cp "$BACKUP_DIR/deploy-api.py" /home/admin958168/_docker/ubuntu/
cp "$BACKUP_DIR/webhook-receiver.py" /home/admin958168/_docker/ubuntu/

# Restart services
systemctl --user restart docker-mgmt-api deploy-api webhook-receiver
Step 5: Verification 
# Check all services

# Check versions updated
